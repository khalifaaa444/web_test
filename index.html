let bleScanner = null; let isScanning = false; async function startScan() { if (isScanning) return; try { updateStatus('Starting BLE scan...'); log('Scanning for BLE broadcasts...', 'info'); document.querySelector('button').disabled = true; document.getElementById('stopBtn').disabled = false; // Check if browser supports BLE scanning if (!navigator.bluetooth || !navigator.bluetooth.requestLEScan) { throw new Error('BLE scanning not supported. Try Chrome 79+ on desktop/Android.'); } // Start BLE scanning (no connection needed) bleScanner = await navigator.bluetooth.requestLEScan({ acceptAllAdvertisements: true, // Accept all BLE devices keepRepeatedDevices: true // Keep showing same devices }); isScanning = true; updateStatus('Scanning for BLE broadcasts...'); log('BLE scan started successfully!', 'success'); // Listen for found devices navigator.bluetooth.addEventListener('advertisementreceived', handleAdvertisement); } catch (error) { console.error('Error:', error); handleError(error); } } function handleAdvertisement(event) { // Device information const deviceName = event.device.name || 'Unknown Device'; const rssi = event.rssi; // Only show ESP32 devices (adjust filter as needed) if (!deviceName.includes('ESP32')) { return; // Skip non-ESP32 devices } // Update status updateStatus(`Found: ${deviceName} (RSSI: ${rssi}dBm)`); // Log the device let logMessage = `${deviceName} - RSSI: ${rssi}dBm`; // Check for manufacturer data (ESP32 BLE beacons usually use this) if (event.manufacturerData) { event.manufacturerData.forEach((data, manufacturerId) => { // Convert manufacturer ID to hex (ESP32 often uses 0xFFFF or 0x02E5) const manuIdHex = manufacturerId.toString(16).toUpperCase().padStart(4, '0'); if (manuIdHex === '02E5' || manuIdHex === 'FFFF') { // Common ESP32 IDs logMessage += ` | ESP32 Data: `; // Convert data to hex string let hexStr = ''; let asciiStr = ''; for (let i = 0; i < data.byteLength; i++) { const byte = data.getUint8(i); hexStr += byte.toString(16).padStart(2, '0').toUpperCase() + ' '; // ASCII representation if (byte >= 32 && byte <= 126) { asciiStr += String.fromCharCode(byte); } else { asciiStr += '.'; } } logMessage += `Hex: ${hexStr.trim()} | ASCII: ${asciiStr}`; } }); } // Log service data if present if (event.serviceData && event.serviceData.size > 0) { event.serviceData.forEach((data, uuid) => { logMessage += ` | Service: ${uuid}`; }); } log(logMessage, 'info', rssi); } function stopScan() { if (bleScanner && isScanning) { bleScanner.stop(); navigator.bluetooth.removeEventListener('advertisementreceived', handleAdvertisement); log('BLE scan stopped.', 'info'); isScanning = false; } document.querySelector('button').disabled = false; document.getElementById('stopBtn').disabled = true; updateStatus('Ready to scan again.'); } // Update error handling function handleError(error) { let errorMsg = error.message; if (error.name === 'SecurityError') { errorMsg = 'Bluetooth permission denied. Please allow access.'; } else if (error.name === 'NotAllowedError') { errorMsg = 'Bluetooth scanning not allowed.'; } else if (error.message.includes('requestLEScan')) { errorMsg = 'BLE scanning not supported in this browser.
' + 'Try Chrome 79+ on desktop or Android.'; } updateStatus(errorMsg, true); log(errorMsg, 'error'); document.querySelector('button').disabled = false; document.getElementById('stopBtn').disabled = true; isScanning = false; }
