#include #include #include #include // Standard UUIDs (same for all devices) #define SERVICE_UUID "180a" #define CHARACTERISTIC_UUID "2a29" // CHANGE THIS FOR EACH DEVICE #define DEVICE_NAME "ESP32-Sensor-1" // Change to "ESP32-Sensor-2", "ESP32-Sensor-3", etc. BLEServer* pServer = NULL; BLECharacteristic* pCharacteristic = NULL; uint8_t testData[] = {0x12, 0x34, 'T', 'E', 'S', 'T'}; class MyServerCallbacks: public BLEServerCallbacks { void onConnect(BLEServer* pServer) { Serial.println("✅ Client connected!"); }; void onDisconnect(BLEServer* pServer) { Serial.println("❌ Client disconnected!"); // Restart advertising BLEDevice::getAdvertising()->start(); } }; void setup() { Serial.begin(115200); delay(1000); Serial.println("\n\n========================================"); Serial.println("Starting ESP32 BLE Server"); Serial.println("Device Name: " DEVICE_NAME); Serial.println("========================================"); // Initialize BLE device with unique name BLEDevice::init(DEVICE_NAME); // Create BLE Server pServer = BLEDevice::createServer(); pServer->setCallbacks(new MyServerCallbacks()); // Create BLE Service BLEService *pService = pServer->createService(SERVICE_UUID); // Create BLE Characteristic pCharacteristic = pService->createCharacteristic( CHARACTERISTIC_UUID, BLECharacteristic::PROPERTY_READ | BLECharacteristic::PROPERTY_NOTIFY ); // Add 2902 descriptor for notifications pCharacteristic->addDescriptor(new BLE2902()); // Set initial value pCharacteristic->setValue(testData, sizeof(testData)); // Start the service pService->start(); // Setup advertising BLEAdvertising *pAdvertising = BLEDevice::getAdvertising(); pAdvertising->addServiceUUID(SERVICE_UUID); pAdvertising->setScanResponse(true); pAdvertising->setMinPreferred(0x06); pAdvertising->setMaxPreferred(0x12); // Start advertising BLEDevice::startAdvertising(); Serial.println("✅ BLE Server started!"); Serial.println("Service UUID: " SERVICE_UUID); Serial.println("Characteristic UUID: " CHARACTERISTIC_UUID); Serial.println("Waiting for client connection..."); printData(); } void printData() { Serial.print("Current Data: "); for (int i = 0; i < sizeof(testData); i++) { Serial.printf("%02X ", testData[i]); } Serial.println(); } void loop() { static uint16_t counter = 0; static unsigned long lastUpdate = 0; // Update every 1000ms if (millis() - lastUpdate >= 1000) { lastUpdate = millis(); // Update the counter bytes testData[0] = (counter >> 8) & 0xFF; testData[1] = counter & 0xFF; // Update characteristic value pCharacteristic->setValue(testData, sizeof(testData)); pCharacteristic->notify(); Serial.print("Update #"); Serial.print(counter); Serial.print(": "); printData(); counter++; } delay(100); }
